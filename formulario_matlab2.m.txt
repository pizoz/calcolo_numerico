%% ========================================================================
%  TEMA 1 - Prova di Laboratorio del 21 Giugno 2024
%  ========================================================================
%  ESERCIZIO 1 (6 punti): matrice pentadiagonale "quasi periodica",
%  determinante, autovalore minimo, fattorizzazione LU, metodo di Jacobi.
%  ESERCIZIO 2 (6 punti): problema di Cauchy, metodo di Heun, bisezione
%  sulla soluzione esatta, integrazione con i trapezi.

%% T1.1) Esercizio 1, punti 1-2: costruzione di A, determinante e autovalore minimo
% A (NxN): diagonale principale = primi N naturali positivi in ORDINE
% DECRESCENTE moltiplicati per 10 (quindi A(1,1)=10N, A(N,N)=10); terza
% sopradiagonale e terza sottodiagonale = -2; b = vettore di soli 1.
N = 20;
A = diag(10*(N:-1:1));                  % diagonale: 200,190,...,10
for i = 1:N-3
    A(i,i+3) = -2;                      % terza sopradiagonale
    A(i+3,i) = -2;                      % terza sottodiagonale
end
b = ones(N,1);

detA = det(A)
% -> atteso: det(A) = 2.3790e+38

lambda_min = min(eig(A))
% -> atteso: lambda_min = 9.8670e+00
% (la matrice ha autovalori tutti reali, essendo "quasi" diagonalmente
%  dominante: elementi diagonali grandi rispetto ai soli due -2 fuori
%  diagonale per riga)

%% T1.2) Esercizio 1, punto 3: fattorizzazione LU e sistema triangolare
[L,U,P] = lu(A);
normL_inf = norm(L,inf)
% -> atteso: ||L||_inf = 1.0501e+00

% Risoluzione di Ax=b tramite la fattorizzazione: P*A = L*U, quindi
% si risolve prima L*y = P*b (sostituzione in avanti), poi U*x = y
% (sostituzione all'indietro)
y = L \ (P*b);
x = U \ y;
normY_inf = norm(y,inf)
normX_inf = norm(x,inf)
% -> atteso: ||y||_inf = 1.0515e+00 , ||x||_inf = 1.0622e-01

%% T1.3) Esercizio 1, punto 5: metodo iterativo di Jacobi
x0 = zeros(N,1);
toll = 1e-6; nitmax = 200;
[xJ, nitJ] = myJacobi(A,b,x0,toll,nitmax);
xe = A\b;                                % soluzione "esatta" di riferimento
err_inf = norm(xJ-xe,inf)/norm(xe,inf);
fprintf('Jacobi: iterazioni = %d\n', nitJ);
fprintf('||xJ-xe||_inf/||xe||_inf = %.4e\n', err_inf);
% -> atteso: iterazioni ~ 6 (converge molto rapidamente, essendo A ben
%    condizionata rispetto al metodo di Jacobi: rho(B_J) piccolo),
%    errore relativo ~ 1.6e-07

%% T1.4) Esercizio 2, punto 1: metodo di Heun sul problema di Cauchy
% y' = -y + e^{-t}(21t^2+5),  0<=t<=1,  y(0)=-1
% soluzione esatta: y(t) = e^{-t}(7t^3+5t-1)
f = @(t,y) -y + exp(-t).*(21*t.^2+5);
yex = @(t) exp(-t).*(7*t.^3+5*t-1);

h = 0.05;
T = (0:h:1)';
[T,Y] = heun(f,T,-1);
Y2 = Y(2)                                % valore in t=0.05 (secondo nodo)
% -> atteso: Y2 = -7.1235e-01

%% T1.5) Esercizio 2, punti 2-4: errore, grafico, bisezione, integrale
err_inf = norm(yex(T)-Y, inf)
% -> atteso: ||y(T)-Y||_inf = 1.4451e-03

figure;
plot(T, yex(T), 'r-', T, Y, 'bo-'); grid on;
legend('soluzione esatta','Heun, h=0.05'); xlabel('t'); ylabel('y');
title('TEMA 1 - Esercizio 2: Heun vs soluzione esatta');

% radice di y(t) (soluzione esatta) in [0,1] con la bisezione
[alpha, nit_bis] = bisezione(yex, 0, 1, 1e-5);
fprintf('alpha = %.4e  (nit bisezione = %d)\n', alpha, nit_bis);
% -> atteso: alpha = 1.9034e-01

% integrale I = int_0^1 y(t) dt: usando i valori approssimati (T,Y) di
% Heun con la formula dei trapezi composita
It = trapz(T,Y);
I  = integral(yex,0,1);
fprintf('I = %.4e , It = %.4e , |I-It| = %.4e\n', I, It, abs(I-It));
% -> atteso: I = 1.4866e+00 , It = 1.4876e+00 , |I-It| = 9.9806e-04


%% ========================================================================
%  TEMA 2 - Prova di Laboratorio del 17 Luglio 2024
%  ========================================================================
%  ESERCIZIO 1 (6 punti): interpolazione polinomiale e spline lineare,
%  confronto trapezi/Simpson con integral.
%  ESERCIZIO 2 (6 punti): Eulero implicito, Newton sulla soluzione
%  esatta confrontato con fzero.

%% T2.1) Esercizio 1, punto 1: interpolazione polinomiale di f
% f(x) = e^{cos(x)} / (x^2+1),  x in [0,1]
f = @(x) exp(cos(x))./(x.^2+1);
a = 0; b = 1;

x10 = linspace(a,b,10);
y10 = f(x10);
p = polyfit(x10,y10,9);                 % polinomio interpolatore (grado 9)
xm = (a+b)/2;
p_xm = polyval(p,xm)
z = linspace(a,b,100);
err_p = max(abs(f(z)-polyval(p,z)))
% -> atteso: p(xm) = 1.9241e+00 , err = 4.9430e-06

%% T2.2) Esercizio 1, punto 2: spline lineare interpolante
s1_xm = interp1(x10,y10,xm,'linear')
err_s1 = max(abs(f(z)-interp1(x10,y10,z,'linear')))
% -> atteso: s1(xm) = 1.9234e+00 , err = 1.2200e-02
% (la spline lineare e' molto meno accurata del polinomio di grado 9 su
%  10 nodi: l'errore e' circa 2500 volte piu' grande)

%% T2.3) Esercizio 1, punto 3: integrale, trapezi (3 nodi) e Simpson (2 sottointervalli)
I = integral(f,a,b)
% -> atteso: I = 1.8889e+00

x3 = linspace(a,b,3);                   % 3 nodi equispaziati: 0, 0.5, 1
IT = trapz(x3, f(x3))
% -> atteso: IT = 1.8562e+00

ICS = (b-a)/6 * ( f(a) + 4*f((a+b)/2) + f(b) );   % Simpson, 2 sottointervalli
% -> atteso: ICS = 1.8788e+00

errT  = abs(IT - I)
errCS = abs(ICS - I)
% -> atteso: errT = 3.2733e-02 , errCS = 1.0101e-02
% (coerente con il grado di precisione: Simpson e' molto piu' accurata
%  dei trapezi a parita' di nodi usati)

%% T2.4) Esercizio 2, punti 1-3: Eulero implicito sul problema di Cauchy
% y' = (y+1)/t + 1,  1<=t<=2,  y(1)=-1
% soluzione esatta: y(t) = t*log(t) - 1
f2  = @(t,y) (y+1)./t + 1;
dfy2 = @(t,y) 1./t;                     % derivata di f2 rispetto a y
yex2 = @(t) t.*log(t) - 1;

h = 0.01;
T = (1:h:2)';
[T,Y] = eulimp(f2,T,-1,dfy2,1e-6,100);
YN = Y(end)
% -> atteso: Y_N = 3.9131e-01

err_inf = norm(yex2(T)-Y, inf)
% -> atteso: ||y(T)-Y||_inf = 5.0125e-03

figure;
plot(T, yex2(T), 'r-', T, Y, 'b--'); grid on;
legend('soluzione esatta','Eulero implicito, h=0.01'); xlabel('t'); ylabel('y');
title('TEMA 2 - Esercizio 2: Eulero implicito vs soluzione esatta');

%% T2.5) Esercizio 2, punto 4: radice della soluzione esatta, Newton vs fzero
g  = @(t) t.*log(t) - 1;
dg = @(t) log(t) + 1;
[alphaNewt, nit] = newton(g, dg, 1, 1e-6, 100);
alpha = fzero(g, 1.5);
fprintf('alphaNewt = %.4e , alpha(fzero) = %.4e\n', alphaNewt, alpha);
% -> atteso: alphaNewt = alpha = 1.7632e+00  (i due metodi coincidono
%    alla precisione richiesta)

%% ========================================================================

%  ESERCIZIO 1 (6 punti): matrice circolante-simile, determinante,
%  condizionamento, raggio spettrale della matrice di iterazione di

%  ESERCIZIO 2 (6 punti): Eulero implicito, confronto integral vs

%% T4.1) Esercizio 1, punti 1: costruzione di A, determinante e condizionamento
% A (NxN): diagonale principale COSTANTE = sqrt(N); prima sopradiagonale
% = 2; seconda sottodiagonale = -2; A(1,N) = A(N,1) = 1.
N = 20;
A = sqrt(N)*eye(N);
for i = 1:N-1
    A(i,i+1) = 2;                       % prima sopradiagonale
end
for i = 1:N-2
    A(i+2,i) = -2;                      % seconda sottodiagonale
end
A(1,N) = 1;
A(N,1) = 1;

detA = det(A)
% -> atteso: det(A) = 1.1442e+12

invA = inv(A);
cond_inf = norm(A,inf) * norm(invA,inf)
% -> atteso: ||A||_inf ||A^{-1}||_inf = 1.6422e+01

%% T4.2) Esercizio 1, punto 2: raggio spettrale della matrice di iterazione di Jacobi
D = diag(diag(A));
BJ = -D \ (A - D);                      % matrice di iterazione di Jacobi
rhoBJ = max(abs(eig(BJ)))
% -> atteso: rho(BJ) = 8.4698e-01
% (essendo rho(BJ) < 1, il metodo di Jacobi converge per questa matrice)

%% T4.3) Esercizio 1, punto 3: metodo di Jacobi
b  = ones(N,1);
x0 = zeros(N,1);
toll = 1e-6; nitmax = 200;
[xJ,nitJ] = myJacobi(A,b,x0,toll,nitmax);
xe = A\b;
err_rel = norm(xJ-xe,inf)/norm(xe,inf);
fprintf('Jacobi: iterazioni = %d , ||xJ-xe||_inf/||xe||_inf = %.4e\n', nitJ, err_rel);
% -> atteso: iterazioni ~ 66 (convergenza piu' lenta rispetto al TEMA 1,
%    coerente con rho(BJ)=0.847 vicino a 1), errore relativo ~ 1.98e-06

%% T4.4) Esercizio 2, punto 1: Eulero implicito sul problema di Cauchy
% y' = y + e^t cos(t),  0<=t<=1,  y(0)=0
% soluzione esatta: y(t) = e^t sin(t)

dfy4 = @(t,y) ones(size(y));            % d/dy[y + e^t cos t] = 1

[T,Y] = eulimp(f4,T,0,dfy4,1e-6,100);

% -> atteso: Ymax = 2.3104e+00

err_inf = norm(yex4(T)-Y, inf)
% -> atteso: ||y(T)-Y||_inf = 2.3070e-02

%% T4.5) Esercizio 2, punto 4: integral vs formula del punto medio composita

% -> atteso: I = 9.0933e-01

m1 = 20; m2 = 40;

function [x, nit] = myJacobi(A, b, x0, toll, nitmax)
% [x,nit] = myJacobi(A,b,x0,toll,nitmax)
% Metodo di Jacobi per la risoluzione di A*x=b.
    D = diag(diag(A));
    R = A - D;
    x = x0;
    nit = 0;
    err = toll + 1;
    while err > toll && nit < nitmax
        xnew = D \ (b - R*x);
        err  = norm(xnew-x, inf);
        x    = xnew;
        nit  = nit + 1;
    end
end

function [x, nit] = bisezione(f, a, b, toll)
% BISEZIONE  Calcola uno zero di f in [a,b] con il metodo di bisezione.
%   [x, nit] = bisezione(f, a, b, toll)
    if f(a)*f(b) > 0
        error('bisezione: f(a) e f(b) devono avere segno discorde');
    end
    a0 = a; b0 = b;
    x   = (a0+b0)/2;
    err = (b0-a0)/2;
    nit = 0;
    while err > toll
        if f(x) == 0
            break
        elseif f(a0)*f(x) < 0
            b0 = x;
        else
            a0 = x;
        end
        x   = (a0+b0)/2;
        err = err/2;
        nit = nit + 1;
    end
end

function [x, nit] = newton(f, df, x0, toll, nitmax)
% NEWTON  Calcola uno zero di f con il metodo di Newton.
%   [x, nit] = newton(f, df, x0, toll, nitmax)
    x   = x0;
    nit = 0;
    dx  = toll + 1;
    while dx > toll && nit < nitmax
        x1  = x - f(x)/df(x);
        dx  = abs(x1 - x);
        x   = x1;
        nit = nit + 1;
    end
end

function [T,U] = eulero(fun,T,y0)
% [T,U] = eulero(fun,T,y0)  - Eulero esplicito, ordine 1.
T = T(:);
N = length(T)-1;
U = zeros(N+1,1);
U(1) = y0;
for n = 1:N
    h = T(n+1)-T(n);
    U(n+1) = U(n) + h*fun(T(n),U(n));
end
end

function [T,U] = eulimp(fun,T,y0,dfy,toll,nitmax)
% [T,U] = eulimp(fun,T,y0,dfy,toll,nitmax) - Eulero implicito, ordine 1,
% assolutamente stabile per ogni h. Risolve F(Y)=0 con Newton ad ogni passo.
T = T(:);
N = length(T)-1;
U = zeros(N+1,1);
U(1) = y0;
for n = 1:N
    h = T(n+1)-T(n);
    Y = U(n);
    err = toll+1; it = 0;
    while err > toll && it < nitmax
        Fy  = Y - U(n) - h*fun(T(n+1),Y);
        dFy = 1 - h*dfy(T(n+1),Y);
        Ynew = Y - Fy/dFy;
        err = abs(Ynew-Y);
        Y = Ynew;
        it = it+1;
    end
    U(n+1) = Y;
end
end

function [T,U] = heun(fun,T,y0)
% [T,U] = heun(fun,T,y0) - metodo di Heun (predictor-corrector), ordine 2.
T = T(:);
N = length(T)-1;
U = zeros(N+1,1);
U(1) = y0;
for n = 1:N
    h  = T(n+1)-T(n);
    k1 = fun(T(n),U(n));
    U(n+1) = U(n) + (h/2)*( k1 + fun(T(n+1), U(n)+h*k1) );
end
end
